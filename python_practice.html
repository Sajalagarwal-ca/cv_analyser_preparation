1. Arrays & Hashing

1.1 Two Sum

AmazonGoogle Array / HashMapEasy

Given nums and target, return the indices of two numbers that add up to target. Exactly one solution exists.

Show Solution 
def two_sum(nums, target):
    seen = {}                       # value -> index
    for i, n in enumerate(nums):
        need = target - n
        if need in seen:
            return [seen[need], i]
        seen[n] = i
    return []
Time: O(n)   Space: O(n)

1.3 Top K Frequent Elements

MetaAmazon Heap / BucketMedium

Return the k most frequent elements.

Show Solution 
from collections import Counter

def top_k_frequent(nums, k):
    freq = Counter(nums)
    # Bucket sort: index = frequency
    buckets = [[] for _ in range(len(nums) + 1)]
    for num, f in freq.items():
        buckets[f].append(num)
    out = []
    for i in range(len(buckets) - 1, 0, -1):
        for n in buckets[i]:
            out.append(n)
            if len(out) == k:
                return out
Time: O(n)   Space: O(n)

1.4 Product of Array Except Self

MetaAmazon Prefix / SuffixMedium

Return an array where out[i] is the product of all elements except nums[i]. No division. O(n) time.

Show Solution 
def product_except_self(nums):
    n = len(nums)
    out = [1] * n
    left = 1
    for i in range(n):
        out[i] = left
        left *= nums[i]
    right = 1
    for i in range(n - 1, -1, -1):
        out[i] *= right
        right *= nums[i]
    return out
Time: O(n)   Space: O(1) extra

def is_valid(s):
    pairs = {')': '(', ']': '[', '}': '{'}
    stack = []
    for c in s:
        if c in '([{':
            stack.append(c)
        else:
            if not stack or stack.pop() != pairs[c]:
                return False
    return not stack
Time: O(n)   Space: O(n)

def length_of_longest_substring(s):
    last = {}
    left = best = 0
    for right, c in enumerate(s):
        if c in last and last[c] >= left:
            left = last[c] + 1
        last[c] = right
        best = max(best, right - left + 1)
    return best
Time: O(n)   Space: O(min(n, charset))

    write = 0

        if nums[read] != 0:
            nums[write], nums[read] = nums[read], nums[write]

def max_area(height):
    l, r = 0, len(height) - 1
    best = 0
    while l < r:
        best = max(best, (r - l) * min(height[l], height[r]))
        if height[l] < height[r]:
            l += 1
        else:
            r -= 1
    return best

Given a list of intervals, merge overlapping ones.

def merge(intervals):
    intervals.sort(key=lambda x: x[0])
    out = [intervals[0]]
    for start, end in intervals[1:]:
        if start <= out[-1][1]:
            out[-1][1] = max(out[-1][1], end)
        else:
            out.append([start, end])
    return out
Time: O(n log n)   Space: O(n)

4.2 Meeting Rooms II (min rooms needed)

AmazonHeapMedium
Show Solution 
import heapq

def min_meeting_rooms(intervals):
    intervals.sort(key=lambda x: x[0])
    heap = []                          # min-heap of end times
    for s, e in intervals:
        if heap and heap[0] <= s:
            heapq.heappop(heap)
        heapq.heappush(heap, e)
    return len(heap)
Time: O(n log n)   Space: O(n)

5.2 Binary Tree Level Order Traversal

MetaGoogle BFSMedium

from collections import deque

def level_order(root):
    if not root: return []
    q, out = deque([root]), []
    while q:
        level = []
        for _ in range(len(q)):
            node = q.popleft()
            level.append(node.val)
            if node.left:  q.append(node.left)
            if node.right: q.append(node.right)
        out.append(level)
    return out
Time: O(n)   Space: O(n)

6.2 Coin Change (min coins)

GoogleDPMedium
Show Solution 
def coin_change(coins, amount):
    INF = amount + 1
    dp = [0] + [INF] * amount
    for a in range(1, amount + 1):
        for c in coins:
            if c <= a:
                dp[a] = min(dp[a], dp[a - c] + 1)
    return dp[amount] if dp[amount] != INF else -1
Time: O(n · amount)   Space: O(amount)

7. Design

7.1 LRU Cache

AmazonMeta OrderedDictMedium
Show Solution 
from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity):
        self.cap = capacity
        self.cache = OrderedDict()

    def get(self, key):
        if key not in self.cache:
            return -1
        self.cache.move_to_end(key)
        return self.cache[key]

    def put(self, key, value):
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        if len(self.cache) > self.cap:
            self.cache.popitem(last=False)
Time: O(1) get/put   Space: O(capacity)

8.2 Month-over-Month % Change (equivalent to your SQL LAG query)

Tiger AnalyticsAmazon PandasMedium

Compute previous-month charges and MoM % change per month.

Show Solution 
import pandas as pd

def mom_change(df: pd.DataFrame) -> pd.DataFrame:
    monthly = (df.groupby('month', as_index=False)['cost']
                 .sum()
                 .rename(columns={'cost': 'charges'})
                 .sort_values('month'))
    monthly['previous_month_charges'] = monthly['charges'].shift(1)
    monthly['mom_pct'] = (
        (monthly['charges'] - monthly['previous_month_charges'])
        / monthly['previous_month_charges'] * 100
    )
    return monthly.sort_values('month', ascending=False)

8.6 Consecutive Login Streaks (≥ 3 days) — pandas version

Same problem you solved in SQL — find users with 3+ consecutive login days, in pandas.

import pandas as pd

def users_with_streak(logins: pd.DataFrame, min_len=3):
    df = (logins.assign(login_date=pd.to_datetime(logins['login_date']))
                .drop_duplicates(['user_id', 'login_date'])
                .sort_values(['user_id', 'login_date']))
    # rank within each user
    df['rn'] = df.groupby('user_id').cumcount()
    df['grp'] = df['login_date'] - pd.to_timedelta(df['rn'], unit='D')
    streaks = (df.groupby(['user_id', 'grp'])
                 .size()
                 .reset_index(name='length'))
    return streaks.loc[streaks['length'] >= min_len, 'user_id'].unique().tolist()
Time: O(n log n)   Space: O(n)

9.2 Decorator: timing a function

AmazonMicrosoft LanguageMedium
Show Solution 
import time
from functools import wraps

def timed(fn):
    @wraps(fn)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        try:
            return fn(*args, **kwargs)
        finally:
            elapsed = time.perf_counter() - start
            print(f"{fn.__name__} took {elapsed:.4f}s")
    return wrapper

@timed
def slow(n):
    return sum(range(n))

9.5 Concurrency: I/O vs CPU-bound

NetflixTiger Analytics LanguageHard

When would you use threading vs multiprocessing vs asyncio?

Show Answer
  • asyncio – best for many concurrent I/O calls (HTTP, DB). Single thread, cooperative.
  • threading – I/O-bound where libs are blocking (files, some DB drivers). GIL limits CPU work.
  • multiprocessing – CPU-bound work (NumPy loops, image processing). Bypasses the GIL by spawning processes.
import asyncio, aiohttp

async def fetch(session, url):
    async with session.get(url) as r:
        return await r.text()

async def main(urls):
    async with aiohttp.ClientSession() as s:
        return await asyncio.gather(*(fetch(s, u) for u in urls))
